/* ========================================================================== */ /* SECCIÓN DE SEGUNDA LENGUA PARA TITULACIÓN */ /* ========================================================================== */ // Catálogo base de Lenguas const LENGUAS_TITULACION = [ { id: 'ING', nombre: 'Inglés para Titulación', docente: 'Irais Irene Texcahua Escobar', total_modulos: 5 }, { id: 'NAH', nombre: 'Náhuatl para Titulación', docente: 'CLEyM (Irais Irene Texcahua Escobar)', total_modulos: 5 } ]; /* OBTENER ESTADO DE SEGUNDA LENGUA PARA UN ALUMNO */ app.get("/lenguas-status/:matricula", async (req, res) => { const matricula = req.params.matricula; try { const resultado = await pool.query( `SELECT * FROM alumno_lenguas WHERE matricula = $1 ORDER BY id ASC`, [matricula] ); res.json({ catalogo: LENGUAS_TITULACION, inscripciones: resultado.rows }); } catch (error) { console.log("ERROR LENGUAS STATUS:", error); res.status(500).json({ catalogo: LENGUAS_TITULACION, inscripciones: [] }); } }); /* INSCRIBIRSE A UN MÓDULO DE LENGUA */ app.post("/inscribir-lengua", async (req, res) => { const { matricula, lengua_codigo, modulo } = req.body; try { // Verificar si ya está en proceso o acreditado ese módulo const existe = await pool.query( "SELECT * FROM alumno_lenguas WHERE matricula = $1 AND lengua_codigo = $2 AND modulo = $3", [matricula, lengua_codigo, modulo] ); if (existe.rows.length > 0) { return res.json({ success: false, mensaje: `Ya estás registrado en el Módulo ${modulo}.` }); } await pool.query( "INSERT INTO alumno_lenguas (matricula, lengua_codigo, modulo, estado) VALUES ($1, $2, $3, 'En proceso')", [matricula, lengua_codigo, parseInt(modulo)] ); res.json({ success: true, mensaje: `✅ Inscrito correctamente al Módulo ${modulo} de ${lengua_codigo === 'ING' ? 'Inglés' : 'Náhuatl'}.` }); } catch (error) { console.log("ERROR INSCRIBIR LENGUA:", error); res.status(500).json({ success: false, mensaje: "Error al inscribir el módulo de lengua." }); } }); /* ACREDITAR MÓDULO DE LENGUA (CLAVE: ESCOBAR2026) */ app.put("/acreditar-lengua", async (req, res) => { const { alumno_lengua_id, clave_profesor } = req.body; try { if (String(clave_profesor).trim() !== "ESCOBAR2026") { return res.json({ success: false, mensaje: "❌ Clave de docente incorrecta. Se requiere autorización de Irais Irene Texcahua Escobar / CLEyM." }); } await pool.query( "UPDATE alumno_lenguas SET estado = 'Acreditada' WHERE id = $1", [alumno_lengua_id] ); res.json({ success: true, mensaje: "🎉 ¡Módulo de Lengua Acreditado exitosamente!" }); } catch (error) { console.log("ERROR ACREDITAR LENGUA:", error); res.status(500).json({ success: false, mensaje: "Error al acreditar el módulo de lengua." }); } }); /* KARDEX COMPLETO (INCLUYE MATERIAS, ACTIVIDADES Y SEGUNDA LENGUA) */ app.get("/kardex/:matricula", async (req, res) => { const matricula = req.params.matricula; try { const materiasRes = await pool.query( `SELECT materias.nombre, materias.creditos, calificaciones.final FROM calificaciones INNER JOIN materias ON calificaciones.materia_id = materias.id WHERE calificaciones.matricula = $1`, [matricula] ); const materias = materiasRes.rows; const actividadesRes = await pool.query( `SELECT catalogo_actividades.nombre, catalogo_actividades.categoria AS tipo, alumno_actividades.estado FROM alumno_actividades INNER JOIN catalogo_actividades ON alumno_actividades.actividad_id = catalogo_actividades.id WHERE alumno_actividades.matricula = $1`, [matricula] ); const actividades = actividadesRes.rows; const lenguasRes = await pool.query( `SELECT lengua_codigo, modulo, estado FROM alumno_lenguas WHERE matricula = $1 ORDER BY lengua_codigo, modulo ASC`, [matricula] ); const lenguas = lenguasRes.rows; let suma = 0, creditos = 0; materias.forEach((m) => { suma += parseFloat(m.final || 0); creditos += parseInt(m.creditos || 0); }); const promedio = materias.length > 0 ? suma / materias.length : 0; res.json({ materias, promedio, creditos, actividades, lenguas }); } catch (error) { console.log("ERROR KARDEX:", error); res.status(500).json({ materias: [], promedio: 0, creditos: 0, actividades: [], lenguas: [] }); } });
